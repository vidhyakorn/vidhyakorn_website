{{/* image-optimize (project override) — same contract as the theme's partial, plus support for a remote image URL. A remote image is downloaded once at build time and then resized exactly like a local one, so cards stay fast and the site never hot-links. A failed download degrades to nothing rather than breaking the build. */}} {{- $page := .page -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $img := "" -}} {{- if or (hasPrefix $src "http://") (hasPrefix $src "https://") -}} {{- $opts := dict "headers" (dict "User-Agent" "Mozilla/5.0 (compatible; Hugo static site build)") -}} {{- with try (resources.GetRemote $src $opts) -}} {{- with .Err -}} {{- warnf "image-optimize: could not fetch %q (%s)" $src . -}} {{- else -}} {{- with .Value -}} {{- if eq .MediaType.MainType "image" -}}{{ $img = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- with $page.Resources.GetMatch $src }}{{ $img = . }}{{ end -}} {{- if not $img -}} {{- with resources.Get ($src | strings.TrimPrefix "/") }}{{ $img = . }}{{ end -}} {{- end -}} {{- end -}} {{- if $img -}} {{- if .fill -}} {{- with try ($img.Fill .fill) -}} {{- with .Err -}}{{ warnf "image-optimize: could not resize %q (%s)" $src . }} {{- else -}}{{ with .Value }}{{ end }}{{- end -}} {{- end -}} {{- else if .process -}} {{- with try ($img.Process .process) -}} {{- with .Err -}}{{ warnf "image-optimize: could not process %q (%s)" $src . }} {{- else -}}{{ with .Value }}{{ end }}{{- end -}} {{- end -}} {{- else -}} {{- end -}} {{- else if not (or (hasPrefix $src "http://") (hasPrefix $src "https://")) -}} {{- /* Not a resource Hugo knows about. If the file really is in static/, serve it; if it isn't, emit nothing so the caller can fall back to a tile rather than showing a broken-image icon. A mistyped or renamed path is the usual cause. */ -}} {{- $rel := $src | strings.TrimPrefix "/" -}} {{- if os.FileExists (printf "static/%s" $rel) -}} {{- else -}} {{- warnf "image: %q was not found in static/ or assets/ — check the path" $src -}} {{- end -}} {{- end -}}
